// Injected into the existing PERF-03A runner only during PERF-03B2 capture.
// Runs the same keyboard/DOM click, focus, typing, results and close contract
// on both control and candidate with fresh Chrome profiles.
async function perf03b2SearchProbe(cdp, network, label) {
  const traffic = network.start();
  const started = Date.now();
  if (label === "cold") {
    await evaluate(cdp, 'document.dispatchEvent(new KeyboardEvent("keydown",{bubbles:true,ctrlKey:true,key:"k"})); true');
  } else {
    await evaluate(cdp, '(() => { const button=document.getElementById("openSearchButton"); if (!button) throw new Error("Missing Search open button"); button.click(); return true; })()');
  }
  const deadline = Date.now() + 30000;
  let ready = null;
  while (Date.now() < deadline) {
    ready = await evaluate(cdp, '(() => {const modal=document.getElementById("searchModal");return {visible:!!modal&&!modal.hidden,focused:document.activeElement?.id==="playerSearchInput",runtime:typeof window.__mflGlobalSearchRuntime?.search==="function"};})()');
    if (ready?.visible && ready?.focused && ready?.runtime) break;
    await delay(50);
  }
  assert.ok(ready?.visible && ready.focused && ready.runtime, label + " first-use search/focus failed: " + JSON.stringify(ready));
  const firstFocusMs = Date.now() - started;
  await evaluate(cdp, '(() => {const x=document.getElementById("playerSearchInput");x.value="roma";x.dispatchEvent(new Event("input",{bubbles:true}));return true;})()');
  await delay(600);
  await network.waitForIdle(30000);
  const typed = await evaluate(cdp, '(() => ({text:document.getElementById("playerSearchInput")?.value,results:document.querySelectorAll("#playerSearchResults > .searchResult").length}))()');
  assert.equal(typed.text, "roma", "Search query changed during input.");
  await evaluate(cdp, '(() => {const x=document.getElementById("closeSearchButton");if(!x) throw new Error("No close button");x.click();return true;})()');
  const hidden = await evaluate(cdp, 'document.getElementById("searchModal").hidden');
  assert.equal(hidden, true, label + " close button did not hide Search.");
  const ms = Date.now() - started;
  network.stop();
  return { firstFocusMs, totalMs: ms, resultCount: typed.results,
    requests: traffic.requestCount, apiRequests: traffic.apiRequestCount, encodedBytes: traffic.bytes };
}
